cmake_minimum_required(VERSION 3.25)
project(pecislav-group-mixer VERSION 0.2.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets)
set(OBS_SOURCE_DIR "" CACHE PATH "OBS source tree matching the target installation")
set(OBS_IMPORT_DIR "" CACHE PATH "Directory with obs.lib and obs-frontend-api.lib")
if(NOT EXISTS "${OBS_SOURCE_DIR}/libobs/obs.h")
  message(FATAL_ERROR "Set OBS_SOURCE_DIR to the OBS 33.0.2 source tree")
endif()
configure_file(src/obsconfig.h.in generated/obsconfig.h @ONLY)
add_library(pecislav-group-mixer MODULE src/plugin.cpp src/mixer.cpp src/model.cpp src/i18n.cpp)
target_include_directories(pecislav-group-mixer PRIVATE
  "${CMAKE_CURRENT_BINARY_DIR}/generated" "${OBS_SOURCE_DIR}/libobs"
  "${OBS_SOURCE_DIR}/frontend/api")
target_link_libraries(pecislav-group-mixer PRIVATE Qt6::Widgets
  "${OBS_IMPORT_DIR}/obs.lib" "${OBS_IMPORT_DIR}/obs-frontend-api.lib")
target_compile_definitions(pecislav-group-mixer PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
if(MSVC)
  target_compile_options(pecislav-group-mixer PRIVATE /utf-8 /W4 /permissive-)
endif()
set_target_properties(pecislav-group-mixer PROPERTIES PREFIX "")
install(TARGETS pecislav-group-mixer LIBRARY DESTINATION bin/64bit RUNTIME DESTINATION bin/64bit)
install(DIRECTORY data/locale/ DESTINATION data/locale FILES_MATCHING PATTERN "*.ini")

include(CTest)
if(BUILD_TESTING)
  add_executable(model-tests tests/model-tests.cpp src/model.cpp)
  target_include_directories(model-tests PRIVATE src)
  target_link_libraries(model-tests PRIVATE Qt6::Widgets)
  if(MSVC)
    target_compile_options(model-tests PRIVATE /utf-8)
  endif()
  add_test(NAME mixer-model COMMAND model-tests)
  add_executable(mixer-smoke tests/mixer-smoke.cpp src/mixer.cpp src/model.cpp src/i18n.cpp)
  target_include_directories(mixer-smoke PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/generated"
    "${OBS_SOURCE_DIR}/libobs" "${OBS_SOURCE_DIR}/frontend/api")
  target_link_libraries(mixer-smoke PRIVATE Qt6::Widgets "${OBS_IMPORT_DIR}/obs.lib"
    "${OBS_IMPORT_DIR}/obs-frontend-api.lib")
  target_compile_definitions(mixer-smoke PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  if(MSVC)
    target_compile_options(mixer-smoke PRIVATE /utf-8)
  endif()
  add_library(obs-test-driver MODULE tests/obs-driver.cpp)
  target_include_directories(obs-test-driver PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated"
    "${OBS_SOURCE_DIR}/libobs" "${OBS_SOURCE_DIR}/frontend/api")
  target_link_libraries(obs-test-driver PRIVATE Qt6::Widgets "${OBS_IMPORT_DIR}/obs.lib"
    "${OBS_IMPORT_DIR}/obs-frontend-api.lib")
  target_compile_definitions(obs-test-driver PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  set_target_properties(obs-test-driver PROPERTIES PREFIX "")
  if(MSVC)
    target_compile_options(obs-test-driver PRIVATE /utf-8)
  endif()
endif()
